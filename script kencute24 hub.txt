-- Utils
local function Char()
	return LocalPlayer.Character
end
local function HRP()
	return Char() and Char():FindFirstChild("HumanoidRootPart")
end
local function Hum()
	return Char() and Char():FindFirstChildOfClass("Humanoid")
end

	CurrentValue = false,

		pcall(function()
			local h = Hum()

			end
		end)

-- ESP (Player xanh / Nextbot đỏ)
local function ClearESP()
	for _,o in pairs(ESPObjs) do pcall(function() o:Destroy() end) end
	ESPObjs = {}
end

local function AddESP(model, color)
	if not model:FindFirstChild("HumanoidRootPart") then return end
	local a = Instance.new("BoxHandleAdornment")
	a.Adornee = model.HumanoidRootPart
	a.Size = Vector3.new(4,6,4)
	a.AlwaysOnTop = true
	a.Transparency = 0.5
	a.ZIndex = 5
	a.Color3 = color
	a.Parent = model
	table.insert(ESPObjs, a)
end

Main:CreateToggle({
	Name = "ESP Player / Nextbot",
	CurrentValue = false,
	Callback = function(v)
		ESPOn = v
		ClearESP()
		if v then
			for _,m in pairs(workspace:GetDescendants()) do
				if m:IsA("Model") and m:FindFirstChild("HumanoidRootPart") then
					if Players:GetPlayerFromCharacter(m) then
						AddESP(m, Color3.fromRGB(0,255,0))
					elseif m.Name:lower():find("bot") then
						AddESP(m, Color3.fromRGB(255,0,0))
					end
				end
			end
		end
	end
})

-- Dash / Slide (Q = Dash, C = Slide)
UIS.InputBegan:Connect(function(i,g)
	if g then return end
	if i.KeyCode == Enum.KeyCode.Q and not DashCD and HRP() then
		DashCD = true
		HRP().Velocity = Camera.CFrame.LookVector * 90
		task.delay(0.6,function() DashCD=false end)
	end
	if i.KeyCode == Enum.KeyCode.C and HRP() then
		HRP().Velocity = Camera.CFrame.LookVector * 60
	end
end)

RunService.Heartbeat:Connect(function()
	pcall(function()
		if HRP() then
			if AutoWin then
				HRP().Velocity = Vector3.new(0,120,0)
				HRP().CFrame += Vector3.new(0,4,0)
			end
			if AutoCandy then
				for _,c in pairs(workspace:GetDescendants()) do
					if c:IsA("BasePart") and c.Name:lower():find("candy") then
						HRP().CFrame = c.CFrame
						task.wait(0.12)
					end
				end
			end
			if SmartEvade then
				for _,b in pairs(workspace:GetDescendants()) do
					if b:IsA("Model") and b.Name:lower():find("bot") and b:FindFirstChild("HumanoidRootPart") then
						if (b.HumanoidRootPart.Position - HRP().Position).Magnitude < 30 then
							HRP().Velocity = (HRP().Position - b.HumanoidRootPart.Position).Unit * 80
						end
					end
				end
			end
		end
	end)
end)
