if game.PlaceId ~= 9872472334 then return end

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local lp = Players.LocalPlayer

local function getChar()
	return lp.Character or lp.CharacterAdded:Wait()
end

local gui = Instance.new("ScreenGui")
gui.Name = "Kencute24_farmcandy"
gui.ResetOnSpawn = false
gui.Parent = lp:WaitForChild("PlayerGui")

local frame = Instance.new("Frame", gui)
frame.Size = UDim2.fromScale(0.22, 0.15)
frame.Position = UDim2.fromScale(0.39, 0.4)
frame.BackgroundColor3 = Color3.fromRGB(25,25,25)
frame.BorderSizePixel = 0
frame.Active = true
frame.Draggable = true
Instance.new("UICorner", frame).CornerRadius = UDim.new(0,18)

local title = Instance.new("TextLabel", frame)
title.Size = UDim2.fromScale(1,0.35)
title.BackgroundTransparency = 1
title.Text = "kencute24 | Farm Candy"
title.TextColor3 = Color3.new(1,1,1)
title.TextScaled = true
title.Font = Enum.Font.GothamBold

local toggle = Instance.new("TextButton", frame)
toggle.Size = UDim2.fromScale(0.8,0.4)
toggle.Position = UDim2.fromScale(0.1,0.5)
toggle.Text = "OFF"
toggle.TextScaled = true
toggle.Font = Enum.Font.GothamBold
toggle.BackgroundColor3 = Color3.fromRGB(170,0,0)
toggle.TextColor3 = Color3.new(1,1,1)
Instance.new("UICorner", toggle).CornerRadius = UDim.new(0,14)

local enabled = false
local oldCFrame
local platform

local function createPlatform()
	if platform then return end
	platform = Instance.new("Part")
	platform.Size = Vector3.new(200,5,200)
	platform.Anchored = true
	platform.CanCollide = true
	platform.Position = Vector3.new(0,10000,0)
	platform.Parent = workspace
end

local function findCandy()
	for _,v in pairs(workspace:GetDescendants()) do
		if v:IsA("BasePart") and string.find(v.Name:lower(),"candy") then
			return v
		end
	end
end

toggle.MouseButton1Click:Connect(function()
	local char = getChar()
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	enabled = not enabled
	if enabled then
		oldCFrame = hrp.CFrame
		toggle.Text = "ON"
		toggle.BackgroundColor3 = Color3.fromRGB(0,170,0)
		createPlatform()
	else
		toggle.Text = "OFF"
		toggle.BackgroundColor3 = Color3.fromRGB(170,0,0)
		if oldCFrame then
			hrp.CFrame = oldCFrame
		end
	end
end)

RunService.Heartbeat:Connect(function()
	if not enabled then return end
	local char = getChar()
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	local candy = findCandy()
	if candy then
		hrp.CFrame = candy.CFrame + Vector3.new(0,3,0)
	else
		hrp.CFrame = CFrame.new(0,10010,0)
	end
end)
